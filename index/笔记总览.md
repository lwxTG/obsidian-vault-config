# 笔记总览

> 卡片视图 · 按目录分组浏览全部笔记。需要表格筛选视图请用 [[笔记总览.base]]。

```dataviewjs
// ---- 按目录分组的笔记 ----
const groups = [
  ['工作笔记 notes/', dv.pages('"notes"').sort(p => p.file.mtime, 'desc')],
  ['归档 archive/', dv.pages('"archive"').sort(p => p.file.mtime, 'desc')],
  ['学习 study/', dv.pages('"study"').sort(p => p.file.mtime, 'desc')],
  ['便签与指南 work/ + guides/', dv.pages('"work" or "archive/guides"').sort(p => p.file.mtime, 'desc')],
  ['收集 clippings/', dv.pages('"clippings"').sort(p => p.file.mtime, 'desc')]
];

// ---- 渲染 ----
const root = dv.container;
root.classList.add('home-board');
let html = '<div class="hb-grid">';
for (const [name, pages] of groups) {
  const items = pages.where(p => p.file.path != dv.current().file.path).limit(10);
  html += `<div class="hb-panel"><div class="hb-title">${name}（${pages.length}）</div><div class="hb-body">`;
  if (items.length === 0) {
    html += '<p class="empty">暂无笔记</p>';
  } else {
    html += items.map(p => {
      return `<div class="card-item"><a href="${p.file.path}" data-href="${p.file.path}" class="internal-link">${p.file.name}</a><div class="card-meta">${dv.date(p.file.mtime).toFormat('MM-dd')}</div></div>`;
    }).join('');
  }
  html += '</div></div>';
}
html += '</div>';
root.innerHTML = html;
```

```dataviewjs
// ---- 最近更新（全库按修改时间 TOP 15）----
const recent = dv.pages('').where(p => !p.file.folder.startsWith('scripts') && !p.file.folder.startsWith('resource') && !p.file.folder.startsWith('.') && p.file.path != dv.current().file.path).sort(p => p.file.mtime, 'desc').limit(15);
```

```dataviewjs
// ---- 附件体检：attachments 目录名与笔记名一致性 ----
(async () => {
  const ATT = 'resource/attachments';
  const listing = await app.vault.adapter.list(ATT);
  const dirs = (listing && listing.folders) || [];
  const mdFiles = app.vault.getMarkdownFiles().filter(f => !f.path.startsWith('.'));
  // 先读全部 md 内容缓存，避免重复 IO
  const contents = new Map();
  for (const f of mdFiles) {
    contents.set(f.path, await app.vault.adapter.read(f.path));
  }
  const rows = [];
  for (const d of dirs) {
    const name = d.split('/').pop();
    const sub = await app.vault.adapter.list(d);
    const fnames = (sub && sub.files) ? sub.files.map(f => f.split('/').pop()) : [];
    let refs = [];
    for (const f of mdFiles) {
      const c = contents.get(f.path);
      if (c.includes(`attachments/${name}/`) || fnames.some(fn => c.includes(fn))) {
        refs.push(f.basename);
      }
    }
    refs = Array.from(new Set(refs));
    const ok = refs.length > 0 && refs.every(r => r === name);
    rows.push([ok ? '✓' : '⚠', name, refs.length > 0 ? refs.join('、') : '（无引用）', refs.length]);
  }
  const root = dv.container;
  root.classList.add('home-board');
  root.innerHTML = `
  <div class="hb-panel">
    <div class="hb-title">附件体检（${dirs.length} 个目录）</div>
    <div class="hb-body">
      ${rows.length === 0 ? '<p class="empty">无附件目录</p>' : `
      <table>
        <tr><th>状态</th><th>目录名</th><th>引用笔记</th><th>引用数</th></tr>
        ${rows.map(r => `<tr><td>${r[0]}</td><td>${r[1]}</td><td>${r[2]}</td><td>${r[3]}</td></tr>`).join('')}
      </table>`}
    </div>
  </div>`;
})();
```
