## 概览

```dataviewjs
// ---- 数据收集 ----
const current = dv.current().file.path;
const all = dv.pages('').where(p => !p.file.folder.startsWith('scripts') && !p.file.folder.startsWith('resource') && p.file.path != current);
const weekAgo = dv.date('today').minus({days: 7});
const recent = all.where(p => p.file.mtime >= weekAgo);
// ---- 热力图：按天统计笔记更新 ----
const today = dv.date('today').startOf('day');
const dayKey = (dt) => dt.toFormat('yyyy-MM-dd');
const dayCounts = new Map();
for (const p of all) {
  const k = dayKey(p.file.mtime);
  dayCounts.set(k, (dayCounts.get(k) || 0) + 1);
}
const WEEKS = 52;
const startMonday = today.minus({ days: today.weekday - 1 }).minus({ weeks: WEEKS });
const weekCols = [];
let totalYear = 0;
const monthLabelAt = {};
for (let w = 0; w <= WEEKS; w++) {
  const col = [];
  for (let d = 0; d < 7; d++) {
    const date = startMonday.plus({ weeks: w, days: d });
    if (date > today) { col.push(null); continue; }
    const count = dayCounts.get(dayKey(date)) || 0;
    totalYear += count;
    if (date.day === 1 && w < WEEKS) monthLabelAt[w] = `${date.month}月`;
    col.push({ date, count });
  }
  weekCols.push(col);
}
const heatLevel = (c) => c === 0 ? 0 : c <= 2 ? 1 : c <= 5 ? 2 : c <= 9 ? 3 : 4;
let heatColsHtml = '';
const monthSpans = [];
for (let w = 0; w < weekCols.length; w++) {
  let cells = '';
  for (const cell of weekCols[w]) {
    cells += cell
      ? `<i class="hb-heat-cell l${heatLevel(cell.count)}" title="${dayKey(cell.date)} · 更新 ${cell.count} 篇"></i>`
      : '<i class="hb-heat-cell ghost"></i>';
  }
  heatColsHtml += `<div class="hb-heat-col">${cells}</div>`;
  monthSpans.push(`<span>${monthLabelAt[w] || ''}</span>`);
}
// 待办任务：只统计看板目录，避免 work 便签测试任务混入
const openTasks = dv.pages('"kanban"').file.tasks.where(t => !t.completed && !t.path.startsWith('.'));
const work = dv.pages('"work"').sort(p => p.file.mtime, 'desc');
const guides = dv.pages('"archive/guides"').sort(p => p.file.name);
const projects = {
  'GNSS / 定位': dv.pages('#project/gnss'),
  '常驻地': dv.pages('#project/residence')
};

// ---- 渲染容器 ----
const root = dv.container;
root.classList.add('home-board');
root.innerHTML = `
<div class="hb-stats">
  <a href="index/笔记总览.md" data-href="index/笔记总览.md" class="internal-link stat-card"><div class="stat-num">${all.length}</div><div class="stat-label">笔记总数</div><span class="stat-arrow">→</span></a>
  <a href="index/笔记总览.md#最近更新" data-href="index/笔记总览.md#最近更新" class="internal-link stat-card"><div class="stat-num">${recent.length}</div><div class="stat-label">近 7 天更新</div><span class="stat-arrow">→</span></a>
  <a href="kanban/个人任务看板.md#收集箱" data-href="kanban/个人任务看板.md#收集箱" class="internal-link stat-card"><div class="stat-num">${openTasks.length}</div><div class="stat-label">未完成任务</div><span class="stat-arrow">→</span></a>
  <a href="index/笔记总览.md" data-href="index/笔记总览.md" class="internal-link stat-card"><div class="stat-num">${work.length}</div><div class="stat-label">工作便签</div><span class="stat-arrow">→</span></a>
</div>
<div class="hb-panel hb-heatmap"><div class="hb-title">笔记热力图<span class="hb-heat-sub">近一年 ${totalYear} 次笔记更新</span></div><div class="hb-heat-wrap"><div class="hb-heat-months">${monthSpans.join('')}</div><div class="hb-heat">${heatColsHtml}</div></div><div class="hb-heat-foot"><div class="hb-heat-legend"><span>少</span><i class="hb-heat-cell"></i><i class="hb-heat-cell l1"></i><i class="hb-heat-cell l2"></i><i class="hb-heat-cell l3"></i><i class="hb-heat-cell l4"></i><span>多</span></div></div></div>
<div class="hb-grid">
  <div class="hb-panel"><div class="hb-title">待办任务</div><div class="hb-body" id="hb-tasks"></div></div>
  <div class="hb-panel"><div class="hb-title">最近更新</div><div class="hb-body" id="hb-recent"></div></div>
  <div class="hb-panel"><div class="hb-title">项目概览</div><div class="hb-body" id="hb-projects"></div></div>
  <div class="hb-panel"><div class="hb-title">便签与指南</div><div class="hb-body" id="hb-work"></div></div>
</div>`;

// ---- 内部链接 HTML 辅助 ----
const linkHtml = (p, suffix = '') => `<li><a href="${p.file.path}" data-href="${p.file.path}" class="internal-link">${p.file.name}</a>${suffix}</li>`;

// ---- 待办任务 ----
if (openTasks.length === 0) {
  root.querySelector('#hb-tasks').innerHTML = '<p class="empty">暂无待办任务</p>';
} else {
  const taskTarget = root.querySelector('#hb-tasks');
  dv.taskList(openTasks.limit(10), false, taskTarget);
  // 保险：部分 dataview 版本忽略容器参数会把任务渲染到块根部，这里统一移入面板
  // 注意：装了 Tasks 插件时任务列表类名是 contains-task-list（而非 task-list），需同时匹配
  const taskListEl = root.querySelector('ul.task-list, ul.contains-task-list');
  if (taskListEl && taskListEl.parentElement !== taskTarget) {
    taskTarget.innerHTML = '';
    taskTarget.appendChild(taskListEl);
  }
}

// ---- 最近更新 ----
root.querySelector('#hb-recent').innerHTML = `<ul>${recent.limit(8).map(p => linkHtml(p, ` <span class="muted">${dv.date(p.file.mtime).toFormat('MM-dd')}</span>`)).join('')}</ul>`;

// ---- 项目概览 ----
let projHtml = '';
for (const [name, pages] of Object.entries(projects)) {
  const items = pages.sort(p => p.file.mtime, 'desc').limit(4);
  projHtml += `<div class="proj-name">${name}</div><ul>${items.map(p => linkHtml(p)).join('')}</ul>`;
}
root.querySelector('#hb-projects').innerHTML = projHtml || '<p class="empty">暂无项目笔记</p>';

// ---- 便签与指南 ----
root.querySelector('#hb-work').innerHTML = `
  <div class="proj-name">工作便签</div>
  <ul>${work.limit(4).map(p => linkHtml(p, ` <span class="muted">${dv.date(p.file.mtime).toFormat('MM-dd')}</span>`)).join('')}</ul>
  <div class="proj-name">指南 / SOP</div>
  <ul>${guides.limit(5).map(p => linkHtml(p)).join('')}</ul>`;
```
